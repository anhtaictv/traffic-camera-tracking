reid_model.py import torch import torchvision.transforms as transforms from PIL import Image import numpy as np class ReIDModel: def __init__(self, model_path='reid_resnet50.pth'): self.device = torch.device('cuda' if torch.cuda.is_available() else 'cpu') # Load ReID model (ResNet50 backbone) from reid.torchmodels.resnet50 import ResNet50 self.model = ResNet50(num_classes=751).to(self.device) checkpoint = torch.load(model_path, map_location=self.device) self.model.load_state_dict(checkpoint['state_dict']) self.model.eval() # Preprocessing self.transform = transforms.Compose([ transforms.Resize((256, 128)), transforms.ToTensor(), transforms.Normalize([0.485, 0.456, 0.406], [0.229, 0.224, 0.225]) ]) def extract_feature(self, image_crop): """Extract feature vector from person crop""" img_pil = Image.fromarray(image_crop) input_tensor = self.transform(img_pil).unsqueeze(0).to(self.device) with torch.no_grad(): features = self.model(input_tensor) features = features.cpu().numpy()[0] features /= np.linalg.norm(features) # Normalize return features def compute_similarity(self, feat1, feat2): """Compute cosine similarity between two features""" return np.dot(feat1, feat2) ``` --- ### 2. **Person Track Manager** ```python # person_manager.py import numpy as np from collections import defaultdict from datetime import datetime class PersonTrackManager: def __init__(self, max_age=30, min_hits=3): self.max_age = max_age self.min_hits = min_hits self.person_db = {} # global_person_id -> PersonInfo self.next_global_id = 1 class PersonInfo: def __init__(self, person_id, first_seen_camera, first_timestamp): self.global_id = person_id self.first_seen_camera = first_seen_camera self.first_timestamp = first_timestamp self.last_seen_time = first_timestamp self.cameras_visited = set([first_seen_camera]) self.recent_features = [] self.track_history = [] def update(self, camera_id, timestamp, bbox, feature_vector): self.last_seen_time = timestamp self.cameras_visited.add(camera_id) self.recent_features.append(feature_vector) self.track_history.append({ 'camera': camera_id, 'timestamp': timestamp, 'bbox': bbox }) # Keep only recent features (last 10) if len(self.recent_features) > 10: self.recent_features.pop(0) def get_average_feature(self): if not self.recent_features: return None return np.mean(self.recent_features, axis=0) def register_new_person(self, camera_id, timestamp, feature_vector): """Register new person globally""" person_info = self.PersonInfo( self.next_global_id, camera_id, timestamp ) person_info.update(camera_id, timestamp, None, feature_vector) self.person_db[self.next_global_id] = person_info self.next_global_id += 1 return person_info.global_id def match_existing_person(self, camera_id, timestamp, feature_vector, threshold=0.7): """Match incoming detection to existing person""" best_match = None best_similarities = [] for pid, info in self.person_db.items(): avg_feat = info.get_average_feature() if avg_feat is None: continue sim = self.compute_cosine_similarity(feature_vector, avg_feat) best_similarities.append((pid, sim)) if sim >= threshold: if best_match is None or sim > best_similarities[-1][1]: best_match = pid # Update matched person if best_match is not None: self.person_db[best_match].update(camera_id, timestamp, None, feature_vector) return best_match return None @staticmethod def compute_cosine_similarity(v1, v2): norm_v1 = v1 / np.linalg.norm(v1) norm_v2 = v2 / np.linalg.norm(v2) return np.dot(norm_v1, norm_v2) def get_person_summary(self, person_id): """Get summary of person movement across cameras""" if person_id not in self.person_db: return None info = self.person_db[person_id] return { 'global_id': info.global_id, 'cameras_visited': list(info.cameras_visited), 'total_cameras': len(info.cameras_visited), 'track_duration_seconds': (info.last_seen_time - info.first_timestamp).total_seconds(), 'recent_locations': info.track_history[-5:] } ``` --- ### 3. **Multi-Camera Tracker Main Class** ```python # cross_camera_tracker.py import cv2 import numpy as np from threading import Thread from queue import Queue from datetime import datetime import json try: from deep_sort_realtime.deepsort_tracker import DeepSortTracker except ImportError: raise ImportError("Install deep-sort-realtime: pip install deep-sort-realtime") from reid_model import ReIDModel from person_manager import PersonTrackManager class CrossCameraTracker: def __init__(self, num_cameras=3, reid_model_path='reid_resnet50.pth'): self.num_cameras = num_cameras self.tracker_manager = PersonTrackManager(max_age=30, min_hits=3) self.reid_model = ReIDModel(reid_model_path) # Run DeepSORT detection detections = tracker.update(frame) processed_frame = frame.copy() for det in detections: x1, y1, w, h, confidence, cls_id = det[:6] bbox = (int(x1), int(y1), int(w), int(h)) # Extract person ROI roi = frame[y1:y1+h, x1:x1+w] roi = cv2.resize(roi, (128, 256)) # Extract ReID feature try: feature = self.reid_model.extract_feature(roi) except Exception as e: print(f"Feature extraction error: {e}") continue # Try to match with existing person timestamp = datetime.now() matched_pid = self.tracker_manager.match_existing_person( camera_id=camera_id, timestamp=timestamp, feature_vector=feature, threshold=self.match_threshold ) # Draw bounding box color = (0, 255, 0) if matched_pid else (255, 0, 0) label = f"P{matched_pid}" if matched_pid else "New" cv2.rectangle(processed_frame, (x1, y1), (x1+w, y1+h), color, 2) cv2.putText(processed_frame, label, (x1, y1-10), cv2.FONT_HERSHEY_SIMPLEX, 0.5, color, 2) # Log movement if matched_pid: print(f"[P{matched_pid}] Camera {camera_id}: {timestamp.strftime('%H:%M:%S')}") return processed_frame def run_all_cameras(self): """Run tracking loop for all cameras""" self.running = True threads = [] for cam_id in self.camera_trackers.keys(): thread = Thread(target=self._run_single_camera_loop, args=(cam_id,), daemon=True) threads.append(thread) thread.start() for t in threads: t.join() def _run_single_camera_loop(self, camera_id): """Loop for single camera""" cap = self.camera_streams[camera_id] while self.running: ret, frame = cap.read() if not ret: break processed = self.process_frame(camera_id, frame) # Show result cv2.imshow(f'Camera {camera_id}', processed) if cv2.waitKey(1) & 0xFF == ord('q'): break cap.release() def stop(self): """Stop all tracking""" self.running = False for cam_id in self.camera_streams: self.camera_streams[cam_id].release() cv2.destroyAllWindows() def export_tracking_data(self, filename='tracking_log.json'): """Export tracking log to JSON""" logs = [] for pid, info in self.tracker_manager.person_db.items(): summary = self.tracker_manager.get_person_summary(pid) if summary: logs.append(summary) with open(filename, 'w') as f: json.dump(logs, f, indent=2, default=str) print(f"Tracking data exported to {filename}") ``` --- ### 4. **Usage Example** ```python # Step 3: Update matched tracks for r_idx, c_idx in zip(row_ind, col_ind): det_idx = c_idx track_id = row_ind[r_idx] detector_id = camera_ids[c_idx] if detector_id not in self.transformers: continue transformer = self.transformers[detector_id] bbox = detections[det_idx] emb = embeddings[det_idx] if track_id in self.tracks: self.tracks[track_id].update(bbox, emb) # Step 4: Handle unmatched detections (new tracks) for det_idx in unmatched_dets: detector_id = camera_ids[det_idx] if detector_id not in self.transformers: continue transformer = self.transformers[detector_id] bbox = detections[det_idx] emb = embeddings[det_idx] new_track = EnhancedTrack( track_id=self.next_id, init_bbox=bbox, transformer=transformer, calib_data=None # Should pass actual calibration data ) new_track.update(bbox, emb) self.tracks[self.next_id] = new_track self.next_id += 1 return list(self.tracks.values()) def _build_cost_matrix(self, detections: List[np.ndarray], embeddings: List[np.ndarray], camera_ids: List[str]) -> np.ndarray: """Build cost matrix considering both IoU and embedding distance""" num_tracks = len(self.tracks) num_detections = len(detections) cost_iou = np.full((num_tracks, num_detections), 1e5) cost_emb = np.full((num_tracks, num_detections), 1e5) # Compute IoU costs for i, track in enumerate(self.tracks.values()): pred_bbox = track.predict() for j, det in enumerate(detections): iou = compute_iou(pred_bbox, det) cost_iou[i, j] = 1.0 - iou # Compute embedding costs for i, track in enumerate(self.tracks.values()): if track.embedding is None: continue for j, emb in enumerate(embeddings): sim = 1.0 - cosine(track.embedding, emb) cost_emb[i, j] = 1.0 - sim # Combine costs (weighted sum) alpha = 0.5 # Weight for embedding vs IoU combined_cost = alpha * cost_emb + (1 - alpha) * cost_iou return combined_cost ``` --- ### 3. **Cross-Camera Re-ID Enhancement** ```python class CrossCameraReID: """Giải quyết vấn đề re-ID across different camera views""" def __init__(self, view_transformation_model: ViewTransformationModel): self.view_model = view_transformation_model self.reid_embeddings: Dict[int, np.ndarray] = {} self.camera_view_features: Dict[str, np.ndarray] = {} def normalize_embedding_across_views(self, embedding: np.ndarray, source_camera: str, target_camera: str) -> np.ndarray: """Adjust embedding to account for viewpoint differences""" # Get view transformation matrix VTM = self.view_model.get_transformation(source_camera, target_camera) # Apply transformation transformed = VTM @ embedding # L2 normalization norm = np.linalg.norm(transformed) if norm > 0: transformed /= norm return transformed def cross_camera_matching(self, track_source: EnhancedTrack, track_target: EnhancedTrack, source_cam: str, target_cam: str) -> float: """Match tracks across different cameras""" if track_source.embedding is None or track_target.embedding is None: return 0.0 # Normalize embeddings src_norm = self.normalize_embedding_across_views( track_source.embedding, source_cam, target_cam ) tgt_norm = track_target.embedding # Already in target view # Cosine similarity similarity = np.dot(src_norm, tgt_norm) return similarity ``` --- ### 4. **Integration Example** ```python #
